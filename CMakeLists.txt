cmake_minimum_required(VERSION 3.21)
project(ImageBinarization VERSION 0.1 LANGUAGES CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc photo)
add_library(binarization_core native/core/Binarizer.cpp)
target_include_directories(binarization_core PUBLIC native/core ${OpenCV_INCLUDE_DIRS})
target_link_libraries(binarization_core PUBLIC ${OpenCV_LIBS})
include(CTest)
if(BUILD_TESTING)
	add_executable(core_tests native/tests/CoreTests.cpp)
	target_link_libraries(core_tests PRIVATE binarization_core)
	add_test(NAME core_tests COMMAND core_tests)
	add_executable(legacy_parity native/tests/LegacyParity.cpp)
	target_link_libraries(legacy_parity PRIVATE binarization_core)
endif()
